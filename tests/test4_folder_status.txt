//Test 1: Detect Folders - iterate all subtools, show index/name/isFolder/subtoolEye/folderEye/isOpened/folderName
[IButton,"ZPlugin:ZBridgeT4:Detect Folders",
    "Iterate all subtools and show folder/visibility status for each.",
    [VarSet,t4Count,[SubToolGetCount]]
    [If,t4Count <= 0,
    [NoteIButton,"Close"]
    [Note,"No subtools loaded.",,1,4737096,,200]
    [Exit]
    ]
    [VarSet,t4Msg,"Idx|Name|isFol|Vis|Folder\n"]
    [VarSet,t4Idx,0]
    [Loop,t4Count,
    [VarSet,t4Status,[SubToolGetStatus,t4Idx]]
    [SubToolSelect,t4Idx]
    [VarSet,t4Name,[IGetTitle,Tool:Item Info]]
    [VarSet,t4Name,[StrExtract,t4Name,0,[StrLength,t4Name]-2]]
    //Folder detection: 0x400 (Closed) or 0x800 (Opened) bit set
    //用嵌套If表达OR逻辑
    [VarSet,t4IsFol,0]
    [If,(t4Status & 1024) == 1024,
    [VarSet,t4IsFol,1]]
    [If,(t4Status & 2048) == 2048,
    [VarSet,t4IsFol,1]]
    [If,(t4Status & 1) == 1,
    [VarSet,t4Vis,1]
    ,
    [VarSet,t4Vis,0]
    ]
    [VarSet,t4Folder,[SubToolGetFolderName,t4Idx]]
    [VarSet,t4Msg,[StrMerge,t4Msg,[Val,t4Idx],"|",t4Name,"|",[Val,t4IsFol],"|",[Val,t4Vis],"|",t4Folder,"\n"]]
    [VarInc,t4Idx]
    ]
    [NoteIButton,"Close"]
    [Note,t4Msg,,1,4737096,,400]
,,1]

//Test 2: Visibility RW - read current visibility, toggle, read, restore, show round-trip
[IButton,"ZPlugin:ZBridgeT4:Visibility RW",
    "Test visibility read/write round-trip on current subtool.",
    [VarSet,t4Idx,[SubToolGetActiveIndex]]
    [VarSet,t4Status,[SubToolGetStatus,t4Idx]]
    [VarSet,t4Orig,t4Status]
    [If,(t4Status & 1) == 1,
    [VarSet,t4Vis,1]
    ,
    [VarSet,t4Vis,0]
    ]
    [VarSet,t4Msg,[StrMerge,"Original status: ",[Val,t4Orig]," visible=",[Val,t4Vis],"\n"]]
    [If,t4Vis == 1,
    [VarSet,t4Flags,t4Status & 2147483646]
    [SubToolSetStatus,t4Idx,t4Flags]
    ,
    [VarSet,t4Flags,t4Status | 1]
    [SubToolSetStatus,t4Idx,t4Flags]
    ]
    [VarSet,t4Status,[SubToolGetStatus,t4Idx]]
    [If,(t4Status & 1) == 1,
    [VarSet,t4Vis,1]
    ,
    [VarSet,t4Vis,0]
    ]
    [VarSet,t4Msg,[StrMerge,t4Msg,"After toggle: status=",[Val,t4Status]," visible=",[Val,t4Vis],"\n"]]
    [SubToolSetStatus,t4Idx,t4Orig]
    [VarSet,t4Status,[SubToolGetStatus,t4Idx]]
    [If,(t4Status & 1) == 1,
    [VarSet,t4Vis,1]
    ,
    [VarSet,t4Vis,0]
    ]
    [VarSet,t4Msg,[StrMerge,t4Msg,"After restore: status=",[Val,t4Status]," visible=",[Val,t4Vis]]]
    [NoteIButton,"Close"]
    [Note,t4Msg,,1,4737096,,300]
,,1]

//Test 3: Manifest 9-Field - write/read/parse 9-field record via Mem
//Fields: colorize|maxSdiv|curSdiv|subtoolEye|folderEye|isFolder|isOpened|folderName|name

    "Test 9-field manifest record write/read/parse round-trip.",
    [MemDelete,t4Mem]
