//~~~EXPORT ROUTINE~~~

//Manifest record: colorize|maxSdiv|curSdiv|subtoolEye|folderEye|isFolder|isOpened|folderName|name|ext\0

//Folder detection: 0x400 (Closed) or 0x800 (Opened) bit set => isFolder=1
//Visibility: subtoolEye=bit0 (subtool own eye), folderEye=bit1 (folder eye, folder heads only)
//Folder state: isOpened=1 if 0x800 bit set, 0 if 0x400 bit set
[RoutineDef,ZBRunExport,

[VarSet,zbToolName,[IGetTitle,Tool:Item Info]]
[If,[StrLength,zbToolName] <= 0,
[NoteIButton,"Close"]
[Note,"No Tool loaded. Please load a Tool first.",,,4737096,,200]
[Exit]
]
[VarSet,zbSubToolCount,[SubToolGetCount]]
[If,zbSubToolCount <= 0,
[NoteIButton,"Close"]
[Note,"No SubTools found to export.",,,4737096,,200]
[Exit]
]
[VarSet,zbSavePath,[FileNameAsk,"GoZ(*.goz)|*.goz||","export.goz","ZBridge Export - Choose location..."]]
[If,[StrLength,zbSavePath] <= 0,[Exit]]
[VarSet,zbExportTempDir,[FileNameExtract,zbSavePath,1]]
[VarSet,zbConfirmText,[StrMerge,"ZBridge Export\n\nTool: ",zbToolName,"\nSubTools: ",zbSubToolCount,"\n\nOutput Folder:\n",zbExportTempDir]][NoteIButton,"Proceed"]
[NoteIButton,"Cancel"]
[VarSet,zbCf,[Note,zbConfirmText,,,4737096,,300]]
[If,zbCf == 2,[Exit]]
[VarSet,zbActiveST,[SubToolGetActiveIndex]]
//0.5.0: duplicate subtool name check - abort before writing any files
[RoutineCall,ZBRunDupCheck]
[If,zbDupFound == 1,
[SubToolSelect,zbActiveST]
[Exit]
]
[VarSet,zbIdx,0]
[VarSet,zbMeshIdx,0]
[MemDelete,zbManMem]
[MemCreate,zbManMem,32768]
[VarSet,zbStgTotal,0]
[VarSet,zbManOffset,0]
[VarSet,zbManBytes,0]
//0.8.0: 图层快照内存块（layers.txt 累积写入，跨子工具；仅含图层子工具时产出数据）
[MemDelete,zbLayerMem]
[MemCreate,zbLayerMem,16384]
[VarSet,zbLayerOffset,0]
[Loop,zbSubToolCount,
[VarSet,zbStatus,[SubToolGetStatus,zbIdx]]
//Folder detection: 0x400 (Closed) or 0x800 (Opened) bit set => isFolder=1
//用嵌套If表达OR逻辑，避免不确定的布尔运算符
[VarSet,zbIsFolder,0]
[If,(zbStatus & 1024) == 1024,
[VarSet,zbIsFolder,1]]
[If,(zbStatus & 2048) == 2048,
[VarSet,zbIsFolder,1]]
[If,zbIsFolder == 1,
[SubToolSelect,zbIdx]
[VarSet,zbStName,[IGetTitle,Tool:Item Info]]
[VarSet,zbStName,[StrExtract,zbStName,0,[StrLength,zbStName]-2]]
//文件夹条目：folderName=文件夹名（用于Import重建命名），name=子工具名（文件夹头名字）
[VarSet,zbFolderName,[SubToolGetFolderName,zbIdx]]
[VarSet,zbIsFolder,1]
//Folder head: subtoolEye=bit0, folderEye=bit1, isOpened from 0x800
[If,(zbStatus & 1) == 1,
[VarSet,zbSubtoolEye,1]
,
[VarSet,zbSubtoolEye,0]
]
[If,(zbStatus & 2) == 2,
[VarSet,zbFolderEye,1]
,
[VarSet,zbFolderEye,0]
]
[If,(zbStatus & 1024) == 1024,
[VarSet,zbIsOpened,0]
,
[VarSet,zbIsOpened,1]
]
,
[SubToolSelect,zbIdx]
[VarSet,zbStName,[IGetTitle,Tool:Item Info]]
[VarSet,zbStName,[StrExtract,zbStName,0,[StrLength,zbStName]-2]]
[VarSet,zbFolderName,[SubToolGetFolderName,zbIdx]]
[VarSet,zbIsFolder,0]
//Normal subtool: subtoolEye=bit0, folderEye always 0, isOpened always 0
[If,(zbStatus & 1) == 1,
[VarSet,zbSubtoolEye,1]
,
[VarSet,zbSubtoolEye,0]
]
[VarSet,zbFolderEye,0]
[VarSet,zbIsOpened,0]
]
[VarSet,zbColorizeState,0]
[VarSet,zbMaxSdiv,0]
[VarSet,zbCurSdiv,0]
//文件夹头也是网格，需要读取colorize/sdiv属性
[VarSet,zbColorizeState,[IGet,Tool:Polypaint:Colorize]]
[If,[IExists,Tool:Geometry:SDiv],
[If,[IsEnabled,Tool:Geometry:SDiv],
[VarSet,zbCurSdiv,[IGet,Tool:Geometry:SDiv]]
[VarSet,zbMaxSdiv,[IGetMax,Tool:Geometry:SDiv]]
[ISet,Tool:Geometry:SDiv,zbMaxSdiv]

//0.9.0: Stager 阶段（此刻仍在最高SDiv且Dynamic预览已关：与主GoZ几何对齐）

[ISet,Tool:Geometry:SDiv,zbCurSdiv]

//0.6.0: 导出完成后恢复Dynamic预览开关（导出前若为开则重新打开）
[RoutineCall,ZBRunDynRestorePreview]
//0.8.0: 雕刻图层快照（含图层子工具：基准网格 + 逐层快照 + 写 layers.txt）
//内部自行切回最高SDiv执行快照并在结束后恢复zbCurSdiv
[RoutineCall,ZBRunLayerSnapshot]
[VarInc,zbMeshIdx]
[VarInc,zbIdx]
]
[MemResize,zbManMem,zbManOffset]
[VarSet,zbManPath,[StrMerge,zbExportTempDir,"/zbridge_manifest.txt"]]

[VarSet,zbLayFilePath,[StrMerge,zbExportTempDir,"/layers.txt"]]

[MemResize,zbLayerMem,zbLayerOffset]
[MemSaveToFile,zbLayerMem,zbLayFilePath,1]
,
[If,[FileExists,zbLayFilePath],
[FileDelete,zbLayFilePath]
]
]
[MemDelete,zbLayerMem]
[SubToolSelect,zbActiveST]
[VarSet,zbNoteText,[StrMerge,"Export Complete!\n\nTool: ",zbToolName,"\nTotal entries: ",zbSubToolCount,"\nMeshes exported: ",zbMeshIdx,"\nSubTools with Stager stages: ",zbStgTotal,"\n\nFolder:\n",zbExportTempDir]]
[NoteIButton,"Close"]
[Note,zbNoteText,,,4737096,,400]
]

//~~~DUPLICATE SUBTOOL NAME CHECK (pre-export)~~~
//Collects all subtool names (same naming as manifest: IGetTitle minus last 2 chars)
//and aborts export if any duplicates are found. Duplicate names would overwrite
//each other's .goz files (goz named after subtool) and corrupt the import.
//Comparison: equal length + StrFind==0 (prefix at position 0) => identical string.
[RoutineDef,ZBRunDupCheck,
[VarSet,zbDupFound,0]
[VarSet,zbIdx,0]
[MemDelete,zbDupMem]
[MemCreate,zbDupMem,16384]
[VarSet,zbDupOffset,0]
//Pass 1: collect all names into memory block (null-terminated records)
[Loop,zbSubToolCount,
[SubToolSelect,zbIdx]
[VarSet,zbStName,[IGetTitle,Tool:Item Info]]
[VarSet,zbStName,[StrExtract,zbStName,0,[StrLength,zbStName]-2]]
[VarSet,zbDupBytes,[MemWriteString,zbDupMem,zbStName,zbDupOffset]]
[VarSet,zbDupOffset,zbDupOffset + zbDupBytes]
[VarInc,zbIdx]
]
//Pass 2: pairwise compare O(n^2) (acceptable for SubTool counts)
[VarSet,zbDupCount,0]
[VarSet,zbDupMsg,""]
[VarSet,zbDupI,0]
[VarSet,zbDupOffA,0]
[Loop,zbSubToolCount-1,
[VarSet,zbDupNameA,""]
[MemReadString,zbDupMem,zbDupNameA,zbDupOffA]
[VarSet,zbDupLenA,[StrLength,zbDupNameA]]
[VarSet,zbDupOffB,zbDupOffA + zbDupLenA + 1]
[VarSet,zbDupJ,zbDupI+1]
[VarSet,zbDupIsNew,0]
[Loop,zbSubToolCount-zbDupI-1,
[VarSet,zbDupNameB,""]
[MemReadString,zbDupMem,zbDupNameB,zbDupOffB]
[VarSet,zbDupLenB,[StrLength,zbDupNameB]]
[If,zbDupLenA == zbDupLenB,
[If,[StrFind,zbDupNameB,zbDupNameA] == 0,
[If,zbDupIsNew == 0,
[VarSet,zbDupIsNew,1]
[VarInc,zbDupCount]
[If,zbDupCount == 1,
[VarSet,zbDupMsg,zbDupNameA]
,
[VarSet,zbDupMsg,[StrMerge,zbDupMsg,"\n",zbDupNameA]]
]
]
]
]
[VarSet,zbDupOffB,zbDupOffB + zbDupLenB + 1]
[VarInc,zbDupJ]
]
[VarSet,zbDupOffA,zbDupOffA + zbDupLenA + 1]
[VarInc,zbDupI]
]
[MemDelete,zbDupMem]
//Abort export if duplicates found
[If,zbDupCount > 0,
[VarSet,zbDupFound,1]
[VarSet,zbNoteText,[StrMerge,"Duplicate SubTool names detected (",zbDupCount,")!\n\nExport aborted.\n\nPlease rename the following SubTools in ZBrush, then export again:\n\n",zbDupMsg]]
[NoteIButton,"Close"]
//duration arg left empty: wait for user to click the button
[Note,zbNoteText,,,4737096,,400]
]
]

//~~~DYNAMIC SUBDIVISION PREVIEW ATTRIBUTES (0.6.0)~~~
//8 attributes from the Tool:Geometry Dynamic Subdivision panel (s.* paths).
//Export: read attrs -> build ext string (key=value;) -> temporarily turn the
//        Dynamic preview OFF so the preview is not baked into the GoZ geometry
//        -> restore it right after export.
//Import: restore attrs AFTER the Subdivision (Reconstruct Subdiv) restore, so
//        the rebuild cannot reset them. Every path is guarded by [IExists] for
//        low-version ZBrush compatibility (missing paths are skipped silently;
//        missing ext keys fall back to defaults 0/0.0).

//Export side: read the 8 attrs + build ext string + turn preview off if ON
[RoutineDef,ZBRunDynRead,
[VarSet,dynActive,0]
[If,[IExists,Tool:Geometry:s.Dynamic],[VarSet,dynActive,[IGet,Tool:Geometry:s.Dynamic]]]
[VarSet,dynFlat,0]
[If,[IExists,Tool:Geometry:s.FlatSubdiv],[VarSet,dynFlat,[IGet,Tool:Geometry:s.FlatSubdiv]]]
[VarSet,dynSmooth,0]
[If,[IExists,Tool:Geometry:s.SmoothSubdiv],[VarSet,dynSmooth,[IGet,Tool:Geometry:s.SmoothSubdiv]]]
[VarSet,dynThick,0]
[If,[IExists,Tool:Geometry:s.Thickness],[VarSet,dynThick,[IGet,Tool:Geometry:s.Thickness]]]
[VarSet,dynSegs,0]
[If,[IExists,Tool:Geometry:s.Segments],[VarSet,dynSegs,[IGet,Tool:Geometry:s.Segments]]]
[VarSet,dynPost,0]
[If,[IExists,Tool:Geometry:s.Post SubDiv],[VarSet,dynPost,[IGet,Tool:Geometry:s.Post SubDiv]]]
[VarSet,dynOffset,0]
[If,[IExists,Tool:Geometry:s.Offset],[VarSet,dynOffset,[IGet,Tool:Geometry:s.Offset]]]
[VarSet,dynSmoothness,0]
[If,[IExists,Tool:Geometry:s.Smoothness],[VarSet,dynSmoothness,[IGet,Tool:Geometry:s.Smoothness]]]
//Build ext string, only for paths that exist (missing key = default on import)
[VarSet,zbExtStr,""]
[If,[IExists,Tool:Geometry:s.Dynamic],[VarSet,zbExtStr,[StrMerge,zbExtStr,"dynActive=",[Val,dynActive],";"]]]
[If,[IExists,Tool:Geometry:s.FlatSubdiv],[VarSet,zbExtStr,[StrMerge,zbExtStr,"dynFlat=",[Val,dynFlat],";"]]]
[If,[IExists,Tool:Geometry:s.SmoothSubdiv],[VarSet,zbExtStr,[StrMerge,zbExtStr,"dynSmooth=",[Val,dynSmooth],";"]]]
[If,[IExists,Tool:Geometry:s.Thickness],[VarSet,zbExtStr,[StrMerge,zbExtStr,"dynThick=",[Val,dynThick],";"]]]
[If,[IExists,Tool:Geometry:s.Segments],[VarSet,zbExtStr,[StrMerge,zbExtStr,"dynSegs=",[Val,dynSegs],";"]]]
[If,[IExists,Tool:Geometry:s.Post SubDiv],[VarSet,zbExtStr,[StrMerge,zbExtStr,"dynPost=",[Val,dynPost],";"]]]
[If,[IExists,Tool:Geometry:s.Offset],[VarSet,zbExtStr,[StrMerge,zbExtStr,"dynOffset=",[Val,dynOffset],";"]]]
[If,[IExists,Tool:Geometry:s.Smoothness],[VarSet,zbExtStr,[StrMerge,zbExtStr,"dynSmoothness=",[Val,dynSmoothness],";"]]]
//Turn Dynamic preview OFF before GoZ export (avoid baking preview into geometry)
[If,dynActive == 1,
[If,[IExists,Tool:Geometry:s.Dynamic],
[IUnPress,Tool:Geometry:s.Dynamic]
]
]
]

//Export side: re-enable Dynamic preview after export (if it was ON before)
[RoutineDef,ZBRunDynRestorePreview,
[If,dynActive == 1,
[If,[IExists,Tool:Geometry:s.Dynamic],
[IPress,Tool:Geometry:s.Dynamic]
]
]
]

//Import side: find "zbExtKey=" inside zbExtStr, store value (up to ';') in zbExtVal
[RoutineDef,ZBRunExtParse,
[VarSet,zbExtFound,0]
[If,[StrLength,zbExtStr] > 0,
[VarSet,zbExtPos,[StrFind,zbExtKey,zbExtStr]]
[If,zbExtPos >= 0,
[VarSet,zbExtEnd,[StrFind,";",zbExtStr,zbExtPos+[StrLength,zbExtKey]]]
[If,zbExtEnd >= 0,

//~~~COLOR TEXTURE REBIND (0.7.0)~~~
//GoZ does not embed image data - only a GoZ_TAG_TEXTURE_MAP_PATH string block.
//The actual texture .tif is auto-written by ZBrush next to the .goz on export,
//named <gozName>_TXTR.tif (GoZ workflow convention). ZScript cannot rename/move
//files, so the file stays next to the goz and the actual name is recorded.
//Export: ZBRunTexDetect (bound?) -> ZBRunTexVerify (find auto file, after export)
//        -> ZBRunTexExtAppend (append hasTexture/textureFile to ext).
//Import: ZBRunTexRestore - File:Texture:Import (verified button tooltip path),
//wait for async import, select the new texture, flip it (Texture:Flpv in the
//Texture palette), assign via TextureMap picker popup, force Texture On.
//0.7.1 revision: Tool:Texture Map:Import popped "Error encountered while importing
//a file" - Texture palette Import + Clone Txtr was used, but Clone Txtr did not bind.
//0.7.2 revision: back to Tool:Texture Map:Import (still popped the error dialog).
//0.7.3 revision: use File:Texture:Import (button tooltip path, user-verified, auto-
//binds to the Tool slot); legacy Texture:Import kept as fallback for old versions.
//0.7.4 revision: script-mode import does NOT auto-bind (manual mode does); Clone Txtr
//is tool->palette so it cannot bind; Flip V is Texture:Flpv, not Tool:Texture Map.
//Full flow documented in the routine comment above ZBRunTexRestore.

//Export side: detect whether the current SubTool has a texture bound
//(TextureMap title non-empty => bound). Also deletes any leftover texture file
//from a previous export so the post-export FileExists check only reflects the
//file written by THIS export.
[RoutineDef,ZBRunTexDetect,
[VarSet,zbHasTexture,0]
[VarSet,zbTexTitle,""]
[If,[IExists,Tool:Texture Map:TextureMap],
[VarSet,zbTexTitle,[IGetTitle,Tool:Texture Map:TextureMap]]
]
[If,[StrLength,zbTexTitle] > 0,
[VarSet,zbHasTexture,1]
]
[If,zbHasTexture == 1,
[VarSet,zbTexPath,[StrMerge,zbExportTempDir,zbStName,"_TXTR.tif"]]
[If,[FileExists,zbTexPath],
[FileDelete,zbTexPath]
]
]
]

//Export side: after GoZ export, find the auto-generated texture file
//Naming follows the GoZ workflow convention: <gozBaseName>_TXTR.tif
//(the goz file is written as <subtoolName>.goz by ZBridge).
[RoutineDef,ZBRunTexVerify,
[VarSet,zbTexFound,0]
[VarSet,zbTexFile,""]
[If,zbHasTexture == 1,
[VarSet,zbTexPath,[StrMerge,zbExportTempDir,zbStName,"_TXTR.tif"]]
[If,[FileExists,zbTexPath],
[VarSet,zbTexFound,1]
[VarSet,zbTexFile,[StrMerge,zbStName,"_TXTR.tif"]]
]
]
]

//Export side: append hasTexture/textureFile keys to the ext string
//(only when a texture file was actually found on disk)
[RoutineDef,ZBRunTexExtAppend,
[If,zbTexFound == 1,
[VarSet,zbExtStr,[StrMerge,zbExtStr,"hasTexture=1;textureFile=",zbTexFile,";"]]
]
]

//Import side: explicit texture rebind for subtools with hasTexture=1.
//Called at the end of Phase 2 (after Dynamic restore). Texture On is forced to 1.
//0.7.4 revision: File:Texture:Import imports the file into the Texture palette
//(no error), but in script mode it does NOT auto-assign to the subtool (manual
//mode does). Flip V lives in the Texture palette (Texture:Flpv), NOT in
//Tool:Texture Map, so the old flip paths never existed. Clone Txtr is useless
//here: per docs it copies tool->palette ("adds it to the Texture palette as the
//active texture"), not palette->tool. Fix: wait for the async import, select the
//new texture via Texture:Item Info, flip it (Texture:Flpv), then assign it to the
//subtool through the TextureMap picker popup (IPress + IPress "PopUp:").
//0.7.5 revision: PopUp click alone only sets the ACTIVE palette texture (user
//observed the left-toolbar Texture got it, not the Tool map). Recorded ZScript
//shows manual binding = [IPress,Texture:<name>] (press palette item, triggers
//auto-apply) + [IPress,Tool:Texture Map:TextureMap]. Loop up to 3 attempts due
//to the known ZBrush "first run does not take effect" quirk.
[RoutineDef,ZBRunTexRestore,
[VarSet,zbHasTexture,0]
[VarSet,zbExtKey,"hasTexture="]
[RoutineCall,ZBRunExtParse]
[If,zbExtFound == 1,
[VarSet,zbHasTexture,[Val,zbExtVal]]
]
[If,zbHasTexture == 1,
[VarSet,zbExtKey,"textureFile="]
[RoutineCall,ZBRunExtParse]
[If,zbExtFound == 1,
[If,[StrLength,zbExtVal] > 0,
[VarSet,zbTexPath,[StrMerge,zbImportDir,zbExtVal]]
[If,[FileExists,zbTexPath],
//0.7.10: 进入导入流程即置位，供 Phase 2 结束后清理面板（仅当确实导入过贴图才清理）
[VarSet,zbTexImported,1]
//0.7.4: 导入前记录面板纹理数量（导入是异步的，需等待数量增加）
[VarSet,zbTexCount,0]
[If,[IExists,Texture:Item Info],
[VarSet,zbTexCount,[IGetMax,Texture:Item Info]]
]
//1. 导入贴图到 Texture 面板（File:Texture:Import 实测有效；Texture:Import 兜底）
[If,[IExists,File:Texture:Import],
[FileNameSetNext,zbTexPath]
[IPress,File:Texture:Import]
,
[If,[IExists,Texture:Import],
[FileNameSetNext,zbTexPath]
[IPress,Texture:Import]
]
]
//2. 等待导入完成（防止异步导入尚未结束就操作）
[If,[IExists,Texture:Item Info],
[Loop,3000,
[If,[IGetMax,Texture:Item Info] > zbTexCount,
[LoopExit]
]
[Delay,0.1]
]
]
//3. 选中刚导入的纹理（最后一个）并读取其标题用于拾取器匹配
[VarSet,zbTexTitle,""]
[If,[IExists,Texture:Current Texture],
[ISet,Texture:Item Info,[IGetMax,Texture:Item Info]]
[VarSet,zbTexTitle,[IGetTitle,Texture:Current Texture]]
,
[If,[IExists,Texture:Item Info],
[ISet,Texture:Item Info,[IGetMax,Texture:Item Info]]
[VarSet,zbTexTitle,[IGetTitle,Texture:Item Info]]
[If,[StrLength,zbTexTitle] > 2,
[VarSet,zbTexTitle,[StrExtract,zbTexTitle,0,[StrLength,zbTexTitle]-2]]
]
]
]
//4. 垂直翻转：Flip V 位于 Texture 面板，脚本路径 Texture:Flpv（录制实证）
[If,[IExists,Texture:Flpv],
[IPress,Texture:Flpv]
,
[If,[IExists,Texture:FlipV],
[IPress,Texture:FlipV]
,
[If,[IExists,Texture:Flip Vertically],
[IPress,Texture:Flip Vertically]
]
]
]
//5. 绑定到子工具（多机制组合，模拟录制脚本的手动选择）
//0.7.5: 录制脚本显示手动绑定 = [IPress,Texture:<名>]（面板列表按下，触发自动贴到
//子工具）+ [IPress,Tool:Texture Map:TextureMap]；拾取器 PopUp 点击作为补充。
//循环最多 3 次尝试（ZBrush 已知怪癖：首次运行常不生效，第二次生效）。
//0.7.6: 绑定前显式重选目标子工具（SubToolSelect,zbIdx），确保拾取器赋值落到正确
//子工具（Tool 贴图槽是每子工具一份）；PopUp 点击改用 IClick（2015 官方规范：
//工具贴图槽提交用 IClick），IPress 保留为兜底。
//0.7.7: PopUp 操作前先 IExists 判空。实测：IClick 提交成功后拾取器随即关闭，紧随
//其后的 IPress 会因 PopUp 项不存在报 "Interface item could not be found"（绑定
//已生效但弹错）。分开判空：IClick 关闭拾取器后 → 第二个 IExists=0 → IPress 自动跳过。
[SubToolSelect,zbIdx]
[If,[StrLength,zbTexTitle] > 0,
[If,[IExists,Tool:Texture Map:TextureMap],
[VarSet,zbTexCheck,""]
[Loop,3,
[If,[StrLength,zbTexCheck] <= 0,
[Delay,0.2]
[IPress,[StrMerge,"Texture:",[Var,zbTexTitle]]]
[IPress,Tool:Texture Map:TextureMap]
[If,[IExists,[StrMerge,"PopUp:",[Var,zbTexTitle]]],
[IClick,[StrMerge,"PopUp:",[Var,zbTexTitle]]]
]
[If,[IExists,[StrMerge,"PopUp:",[Var,zbTexTitle]]],
[IPress,[StrMerge,"PopUp:",[Var,zbTexTitle]]]
]
[VarSet,zbTexCheck,""]
[If,[IExists,Tool:Texture Map:TextureMap],
[VarSet,zbTexCheck,[IGetTitle,Tool:Texture Map:TextureMap]]
]
]
]
]
]
//6. 确保 Texture On
[If,[IExists,Tool:Texture Map:Texture On],
[ISet,Tool:Texture Map:Texture On,1]
]
,
[VarInc,zbImportErrors]
]
]
]
]
]

//Tool:Layers 图层快照导出（方案B：平级命名 <stName>_LayerBase.goz / <stName>_Layer<i>.goz）
//机制定案（probe4f/4h/4k 实测）:
//  层数 = ISet(Layers Scrollbar,0,255) 后 IGetSecondary = N(0-based)，总层数 N+1；无 Layers 面板 => 0

//layers.txt 记录: subtoolName|layerIndex|layerName|intensity|visible|enabled|vertexCount\0
//  layerIndex 0=原始栈底层（导入按此顺序 New，重建后栈序一致）
//  enabled/vertexCount 无 ZScript 读取途径，暂记 0 占位
//每次导出循环前 zbLayerMem 已清空重建；zbLayerOffset 跨子工具累积
//（与 manifest 相同的 null-terminated records 格式）

//主控：检测 -> 记录 -> 导出 -> 恢复（无图层子工具整体跳过，行为零变化）
[RoutineDef,ZBRunLayerSnapshot,
[RoutineCall,ZBRunLayerDetect]
[If,zbLayerCount > 0,
[VarSet,zbLayerRestOff,zbLayerOffset]
//主循环此刻已把 SDiv 恢复回 curSdiv；快照期间须保持在最高细分（顶点对齐）

[ISet,Tool:Geometry:SDiv,zbMaxSdiv]

//快照期间临时关闭 Dynamic 预览（主循环已执行 DynRestorePreview，若原开则已重开；

[If,dynActive == 1,
[If,[IExists,Tool:Geometry:s.Dynamic],
[IUnPress,Tool:Geometry:s.Dynamic]
]

[RoutineCall,ZBRunLayerBuildMem]

[RoutineCall,ZBRunLayerRestoreMem]
//恢复 Dynamic 预览与 SDiv
[If,dynActive == 1,
[If,[IExists,Tool:Geometry:s.Dynamic],

[If,zbMaxSdiv > 0,
[ISet,Tool:Geometry:SDiv,zbCurSdiv]

[RoutineDef,ZBRunLayerBuildMem,

[VarSet,zbLayerI,0]
[Loop,zbLayerCount,
[VarSet,zbLayerPath,[IGetTitle,Tool:Layers:Layer Intensity,1]]
[VarSet,zbLayerName,[IGetTitle,Tool:Layers:Layer Intensity]]

//主流程：逐层 归零->New(录制)->导入快照->退录制->命名，最后恢复强度+可见性

[RoutineDef,ZBRunLayerRestoreImport,
[If,zbLayerImpOk == 1,
[RoutineCall,ZBRunLayerImpMatch]
[If,zbLayerImpCount > 0,
//1. 切到最高细分级别（快照均在最高SDiv导出，顶点一一对齐）
[VarSet,zbLayerMaxS,0]

[If,zbLayerFound == 1,
[VarSet,zbLayGozPath,[StrMerge,zbImportDir,zbStName,"_Layer",[Val,zbLayerImpK],".goz"]]

[VarSet,zbLayerI,0]

[If,zbLayerRecBit == 1,

[If,[IsEnabled,Tool:Layers:Layer Intensity],
[VarSet,zbLayerGone,0]

[VarSet,zbLayerGone,1]
[VarSet,zbLayerCount,zbLayerDbg]
[VarInc,zbImportErrors]

//退录制：bit0 确认录制态才点击（此时层名为默认名唯一，拼路径安全）
[VarSet,zbLayerPath,[StrMerge,"Tool:Layers:",[IGetTitle,Tool:Layers:Layer Intensity]]]
[VarSet,zbLayerMod,[IModGet,zbLayerPath]]

[VarSet,zbLayerW,[IWidth,zbLayerPath]]
[VarSet,zbLayerH,[IHeight,zbLayerPath]]
[IClick,zbLayerPath,zbLayerW-25,zbLayerH/2]

//命名（ZFU；任何环节失败降级跳过，不影响形变还原）
[If,[StrLength,zbLayerName] > 0,
[If,[StrLength,dllPath] > 0,
[If,[IExists,Tool:Layers:Rename],
[If,[IsEnabled,Tool:Layers:Rename],
[RoutineCall,ZFU_RenameSetNext,zbLayerName]
[IPress,Tool:Layers:Rename]
]

[VarInc,zbImportErrors]
]
,
[VarSet,zbLayerCount,zbLayerDbg]
[VarInc,zbImportErrors]
]
,
[VarSet,zbLayerCount,zbLayerDbg]
[VarInc,zbImportErrors]
]
,
[VarInc,zbImportErrors]
]
,
[VarInc,zbImportErrors]
]
[VarInc,zbLayerImpK]

[VarSet,zbLayerJ,0]

[VarSet,zbLayerW,[IWidth,zbLayerPath]]
[VarSet,zbLayerH,[IHeight,zbLayerPath]]
[IClick,zbLayerPath,zbLayerW-5,zbLayerH/2]

[IPress,Tool:Layers:SelectDown]
]
[VarInc,zbLayerJ]
]

//4. 切回原显示细分级别
[If,zbLayerMaxS > 0,
[If,[IExists,Tool:Geometry:SDiv],
[If,[IsEnabled,Tool:Geometry:SDiv],
[ISet,Tool:Geometry:SDiv,zbCurSdiv]
]
]
]
]
]

//~~~STAGER STAGES (0.9.0)~~~
//Tool:Geometry:Stager 为每个子工具保存 Home Stage / Target Stage 两个世界空间摆位，
//GoZ 只带当前摆位，另一阶段会丢失。做法与图层快照同思路（GoZ 负责几何，插件负责状态）：
//  导出：读取 Home/Target 指示；两阶段都存在时（Switch Stage 可用）
//        Switch Stage -> 导出 <name>_StageAlt.goz -> Switch Stage 切回 -> 回写主 GoZ。
//        ext 追加：stg=1;stgHome=h;stgTarget=t;stgCur=0;stgAlt=a;stgCount=n;
//  导入：当前网格 = 导出时的当前摆位 C；存为 Home -> 导入 StageAlt（同拓扑，
//        仅改顶点、保留细分）-> 存为 Target -> Switch Stage 回到 C。
//控件判定（ZBrush 2022.0.5 / 2025.0.1 实测一致，tests/Stager_probe.txt）：
//  Home/Target 指示 = IGet(Tool:Geometry:Home Stage / Target Stage)，0=未存，1=已存
//  两阶段齐全       = IsEnabled(Tool:Geometry:Switch Stage)
//  当前阶段         = 无法读取（Switch Stage 的 IGet 恒为 0，IModGet/IGetSecondary 不支持）
//                     => 一律把导出时的当前摆位记为 Home（stgCur=0）。导出时若停在 Target，
//                     迁移后两个摆位都在、Switch 照常往返，只是 Home/Target 名称互换。
//全部路径 IExists 防护：2018 之前无 Stager 的版本整体跳过，ext 中的 stg* 键被忽略。
//限制：Interpolate 生成的中间阶段不单独导出（导入后可用 Stages Count + Interpolate 重新生成）；
//      含雕刻图层并改用 _LayerBase 导入的子工具跳过阶段重建（几何基准不同），计入 zbStgSkipped。

//导出侧：调用时机 = 主 GoZ 导出之后、manifest 行写入之前（最高SDiv、Dynamic 预览已关）
[RoutineDef,ZBRunStgSnapshot,
[VarSet,zbStgHas,0]
[VarSet,zbStgHome,0]
[VarSet,zbStgTarget,0]
[VarSet,zbStgCur,0]
[VarSet,zbStgCount,0]
[VarSet,zbStgAltOk,0]
[If,[IExists,Tool:Geometry:Home Stage],
[If,[IGet,Tool:Geometry:Home Stage] != 0,
[VarSet,zbStgHome,1]
]
]
[If,[IExists,Tool:Geometry:Target Stage],
[If,[IGet,Tool:Geometry:Target Stage] != 0,
[VarSet,zbStgTarget,1]
]
]
[If,[IExists,Tool:Geometry:Stages Count],
[VarSet,zbStgCount,[IGet,Tool:Geometry:Stages Count]]
]
//含雕刻图层的子工具不做 Switch 往返（导入侧本就跳过；往返后回写主网格的 Import 可能清空图层）
[RoutineCall,ZBRunLayerDetect]
[If,[IExists,Tool:Geometry:Switch Stage],
[If,[IsEnabled,Tool:Geometry:Switch Stage],
[VarSet,zbStgHas,1]
[If,zbLayerCount == 0,
//切到另一阶段 -> 导出快照 -> 切回（同拓扑，仅顶点位置不同）
[VarSet,zbStgPath,[StrMerge,zbExportTempDir,"/",zbStName,"_StageAlt.goz"]]
[If,[FileExists,zbStgPath],
[FileDelete,zbStgPath]
]
[IPress,Tool:Geometry:Switch Stage]
[FileNameSetNext,zbStgPath]
[IPress,Tool:Export]
[IPress,Tool:Geometry:Switch Stage]
[If,[FileExists,zbStgPath],
[VarSet,zbStgAltOk,1]
]
//实测：Switch Stage 总是落到已存阶段，存阶段后又移动、未再存的摆位会丢；
//把刚导出的主 GoZ（导出前的当前网格）导回去，保证导出不改动用户场景
[If,[FileExists,zbGozPath],
[FileNameSetNext,zbGozPath]
[IPress,Tool:Import]
]
]
]
]
//任一阶段存在才写键（无 Stager 的子工具 ext 不变，旧版导入完全无感）
[If,(zbStgHas + zbStgHome + zbStgTarget) > 0,
[VarSet,zbExtStr,[StrMerge,zbExtStr,"stg=1;stgHome=",[Val,zbStgHome],";stgTarget=",[Val,zbStgTarget],";"]]
[VarSet,zbExtStr,[StrMerge,zbExtStr,"stgCur=",[Val,zbStgCur],";stgAlt=",[Val,zbStgAltOk],";stgCount=",[Val,zbStgCount],";"]]
[VarInc,zbStgTotal]
]
]

//导入侧：Phase 2 内、Reconstruct Subdiv 之前调用（网格仍是主 GoZ 的最高级网格，无细分级别）
//依赖调用方先解析好 zbStName / zbExtStr
[RoutineDef,ZBRunStgRestore,
[VarSet,zbExtKey,"stg="]
[RoutineCall,ZBRunExtParse]
[If,zbExtFound == 1,
[If,[IExists,Tool:Geometry:Home Stage],
//读取导出时的阶段状态
[VarSet,zbStgHome,0]
[VarSet,zbExtKey,"stgHome="]
[RoutineCall,ZBRunExtParse]
[If,zbExtFound == 1,[VarSet,zbStgHome,[Val,zbExtVal]]]
[VarSet,zbStgTarget,0]
[VarSet,zbExtKey,"stgTarget="]
[RoutineCall,ZBRunExtParse]
[If,zbExtFound == 1,[VarSet,zbStgTarget,[Val,zbExtVal]]]
[VarSet,zbStgCur,0]
[VarSet,zbExtKey,"stgCur="]
[RoutineCall,ZBRunExtParse]
[If,zbExtFound == 1,[VarSet,zbStgCur,[Val,zbExtVal]]]
[VarSet,zbStgAltOk,0]
[VarSet,zbExtKey,"stgAlt="]
[RoutineCall,ZBRunExtParse]
[If,zbExtFound == 1,[VarSet,zbStgAltOk,[Val,zbExtVal]]]
[VarSet,zbStgCount,0]
[VarSet,zbExtKey,"stgCount="]
[RoutineCall,ZBRunExtParse]
[If,zbExtFound == 1,[VarSet,zbStgCount,[Val,zbExtVal]]]
//含图层且主网格来自 _LayerBase 的子工具：几何基准不同，跳过
[VarSet,zbLayerBaseUsed,0]
[If,zbLayerImpOk == 1,
[RoutineCall,ZBRunLayerImpMatch]
[If,zbLayerImpCount > 0,
[If,[FileExists,[StrMerge,zbImportDir,zbStName,"_LayerBase.goz"]],
[VarSet,zbLayerBaseUsed,1]
]
]
]
[If,zbLayerBaseUsed == 1,
[VarInc,zbStgSkipped]
,
[VarSet,zbStgPath,[StrMerge,zbImportDir,zbStName,"_StageAlt.goz"]]
[If,zbStgAltOk == 1,
[If,[FileExists,zbStgPath],
//1. 当前网格 = 导出时的当前阶段 C -> 存为对应阶段
[If,zbStgCur == 0,
[IPress,Tool:Geometry:Home Stage]
,
[IPress,Tool:Geometry:Target Stage]
]
//2. 导入另一阶段快照（同拓扑：只更新顶点位置，保留细分历史）
[FileNameSetNext,zbStgPath]
[IPress,Tool:Import]
//导入会把子工具名改成 goz 名 -> 改回 manifest 名（ZFU 不可用时仅残留后缀）
[If,[StrLength,dllPath] > 0,
[RoutineCall,ZFU_RenameCurrentSubTool,zbStName]
]
//3. 存为另一阶段 -> 切回 C
[If,zbStgCur == 0,
[IPress,Tool:Geometry:Target Stage]
,
[IPress,Tool:Geometry:Home Stage]
]
[If,[IExists,Tool:Geometry:Switch Stage],
[If,[IsEnabled,Tool:Geometry:Switch Stage],
[IPress,Tool:Geometry:Switch Stage]
]
]
[VarInc,zbStgTotal]
,
[VarInc,zbImportErrors]
]
,
//只存了单个阶段：该阶段即当前摆位，直接存回
[If,zbStgHome == 1,
[IPress,Tool:Geometry:Home Stage]
]
[If,zbStgTarget == 1,
[IPress,Tool:Geometry:Target Stage]
]
[VarInc,zbStgTotal]
]
//Stages Count（Interpolate 中间阶段数）
[If,zbStgCount > 0,
[If,[IExists,Tool:Geometry:Stages Count],
[ISet,Tool:Geometry:Stages Count,zbStgCount]
]
]
]
]
]
]

[VarSet,zbCurSdiv,[Val,zbCurSdivStr]]
[VarSet,zbSubEyeStr,[StrExtract,zbManLine,zbPipePos3+1,zbPipePos4-1]]
[VarSet,zbFolEyeStr,[StrExtract,zbManLine,zbPipePos4+1,zbPipePos5-1]]
[VarSet,zbFolStr,[StrExtract,zbManLine,zbPipePos5+1,zbPipePos6-1]]
//Restore Colorize
[If,[IExists,Tool:Polypaint:Colorize],
[If,[Val,zbPpFlag] == 1,
[ISet,Tool:Polypaint:Colorize,1]
,
[ISet,Tool:Polypaint:Colorize,0]
]
]
//0.9.0: Stager 阶段重建——必须在 Reconstruct Subdiv 之前（ZBrush 2022.0.5 实测）：
//此刻网格 = 主 GoZ 的最高级网格，顶点顺序与 _StageAlt.goz 一致，Import 只更新顶点；
//Reconstruct 之后顶点重排，Import 会整体替换网格（细分级别与阶段一起丢失）。
//ZBRunStgRestore 需要 zbStName / zbExtStr，此处先行解析（Phase 2 后段会再解析一次，结果相同）
[If,zbPipePos9 >= 0,
[VarSet,zbStName,[StrExtract,zbManLine,zbPipePos8+1,zbPipePos9-1]]
[VarSet,zbExtStr,[StrExtract,zbManLine,zbPipePos9+1,[StrLength,zbManLine]-1]]
[RoutineCall,ZBRunStgRestore]
]
//Restore SubDiv
[If,zbSdivFlag == 0,
,
[If,[IExists,Tool:Geometry:Reconstruct Subdiv],
[VarSet,zbMaxSdiv,[Val,zbSdivFlag]]
[VarSet,zbMaxSdiv,zbMaxSdiv - 1]
[If,zbMaxSdiv > 0,
[VarSet,zbTempName,0]
[Loop,zbMaxSdiv,
[If,[IsEnabled,Tool:Geometry:Reconstruct Subdiv],
[IPress,Tool:Geometry:Reconstruct Subdiv]
,
[LoopExit]
]
[VarInc,zbTempName]
]
]
[If,[IExists,Tool:Geometry:SDiv],
[If,[IsEnabled,Tool:Geometry:SDiv],
[ISet,Tool:Geometry:SDiv,zbCurSdiv]
]
]
]
]
//Restore Visibility
//Folders: effective visibility = folderEye (bit1)
//Normal subtools: effective visibility = subtoolEye (bit0)
[VarSet,zbStatus,[SubToolGetStatus,zbIdx]]
[If,[Val,zbFolStr] == 1,
    [If,[Val,zbFolEyeStr] == 1,
        [If,(zbStatus & 1) == 0,
            [VarSet,zbStatus,zbStatus + 1]
        ]
    ,
        [If,(zbStatus & 1) == 1,
            [VarSet,zbStatus,zbStatus - 1]
        ]
    ]
,
    [If,[Val,zbSubEyeStr] == 1,
        [If,(zbStatus & 1) == 0,
            [VarSet,zbStatus,zbStatus + 1]
        ]
    ,
        [If,(zbStatus & 1) == 1,
            [VarSet,zbStatus,zbStatus - 1]
        ]
    ]
]
[SubToolSetStatus,zbIdx,zbStatus]
//0.6.0: 动态细分8属性恢复（必须在SubDiv恢复之后，避免被Reconstruct Subdiv重置）
[RoutineCall,ZBRunDynRestore]
//0.7.0: 颜色贴图显式重绑（Phase 2 末尾，紧接动态细分恢复之后）
[RoutineCall,ZBRunTexRestore]
//0.8.0 T4: Phase 2 循环本身不设置 zbStName（残留=Phase 1 末条记录），必须现解析
[If,zbPipePos9 >= 0,
[VarSet,zbStName,[StrExtract,zbManLine,zbPipePos8+1,zbPipePos9-1]]
,
[VarSet,zbStName,[StrExtract,zbManLine,zbPipePos8+1,[StrLength,zbManLine]-1]]
]
//0.8.0 T4: 图层还原（顺序定案：细分已还原 -> 切最高SDiv -> 录制导入 -> 切回zbCurSdiv）
[RoutineCall,ZBRunLayerRestoreImport]
[VarInc,zbIdx]
]
//0.7.10: 所有贴图载入完成后，将 Texture 面板当前纹理置空（用户诉求：左侧面板不残留贴图）。
//说明：0.7.8 的失败根因是移除 ISet 导致绑定静默跳过（zbTexTitle 空），Remove 本身对
//Tool 槽绑定的影响尚未单独验证；本版保留 ISet，单独验证 Remove：
//若 Tool 槽绑定保持 → 方案成立；若绑定丢失 → 放弃本方案，转 marcus 直接导入法（远期）。
[If,zbTexImported == 1,
[If,[IExists,Texture:Remove],
[IPress,Texture:Remove]
]
]
//Phase 3: Rebuild folder structure (New Folder + MoveUp)
[VarDef,zbFolManIdx,0]
[VarDef,zbFolEyeVal,0]
[VarDef,zbFolOpenVal,0]
[VarDef,zbSubEyeVal,0]
[VarDef,zbFolCursor,0]
[VarDef,zbContentCnt,0]
[VarDef,zbContOff,0]
//Count folders in manifest
[VarSet,zbFolderTotal,0]
[VarSet,zbManOffset,0]
[VarSet,zbIdx,0]
[Loop,zbGozCount,
    [VarSet,zbManLine,""]
    [MemReadString,zbManMem,zbManLine,zbManOffset]
    [VarSet,zbManBytes,[StrLength,zbManLine]+1]
    [RoutineCall,ZBRunManParse]
    [VarSet,zbFolStr,[StrExtract,zbManLine,zbPipePos5+1,zbPipePos6-1]]
    [If,[Val,zbFolStr]==1,
        [VarInc,zbFolderTotal]
    ]
    [VarSet,zbManOffset,zbManOffset+zbManBytes]
    [VarInc,zbIdx]
]
//Process folders bottom-to-top
[If,zbFolderTotal > 0,
    [VarSet,zbFolCursor,zbFolderTotal]
    [Loop,zbFolderTotal,
        [VarDec,zbFolCursor]
        //Find this folder's entry in manifest
        [VarSet,zbManOffset,0]
        [VarSet,zbIdx,0]
        [VarSet,zbContOff,0]
        [Loop,zbGozCount,
            [VarSet,zbManLine,""]
            [MemReadString,zbManMem,zbManLine,zbManOffset]
            [VarSet,zbManBytes,[StrLength,zbManLine]+1]
            [RoutineCall,ZBRunManParse]
            [VarSet,zbFolStr,[StrExtract,zbManLine,zbPipePos5+1,zbPipePos6-1]]
            [If,[Val,zbFolStr]==1,
                [If,zbContOff==zbFolCursor,
                    [VarSet,zbFolManIdx,zbIdx]
                    [VarSet,zbSubEyeVal,[Val,[StrExtract,zbManLine,zbPipePos3+1,zbPipePos4-1]]]
                    [VarSet,zbFolEyeVal,[Val,[StrExtract,zbManLine,zbPipePos4+1,zbPipePos5-1]]]
                    [VarSet,zbFolOpenVal,[Val,[StrExtract,zbManLine,zbPipePos6+1,zbPipePos7-1]]]
                    [VarSet,zbFolName,[StrExtract,zbManLine,zbPipePos7+1,zbPipePos8-1]]
                    [LoopExit]
                ]
                [VarInc,zbContOff]
            ]
            [VarSet,zbManOffset,zbManOffset+zbManBytes]
            [VarInc,zbIdx]
        ]
        //Current subtool index = manifest index (bottom-to-top: no shift from folders above)
        [VarSet,zbCurSubIdx,zbFolManIdx]
        //Count content subtools (scan forward until next folder or end)
        [VarSet,zbContentCnt,0]
        [VarSet,zbManOffset,0]
        [VarSet,zbIdx,0]
        [Loop,zbFolManIdx+1,
            [VarSet,zbManLine,""]
            [MemReadString,zbManMem,zbManLine,zbManOffset]
            [VarSet,zbManBytes,[StrLength,zbManLine]+1]
            [VarSet,zbManOffset,zbManOffset+zbManBytes]
            [VarInc,zbIdx]
        ]
        [Loop,zbGozCount-zbFolManIdx-1,
            [VarSet,zbManLine,""]
            [MemReadString,zbManMem,zbManLine,zbManOffset]
            [VarSet,zbManBytes,[StrLength,zbManLine]+1]
            [RoutineCall,ZBRunManParse]
            [VarSet,zbFolStr,[StrExtract,zbManLine,zbPipePos5+1,zbPipePos6-1]]
            [If,[Val,zbFolStr]==1,
                [LoopExit]
            ]
            //Check zbFolNameStr: consecutive pipes (empty field) = standalone subtool → stop
            [If,zbPipePos8 - zbPipePos7 == 1,
                [LoopExit]
            ]
            [VarInc,zbContentCnt]
            [VarSet,zbManOffset,zbManOffset+zbManBytes]
        ]
        //Create folder (wraps selected subtool with pre-set name)
        [SubToolSelect,zbCurSubIdx]
        //Pre-set folder name to avoid rename dialog
        [If,[StrLength,dllPath] > 0,
            [If,[FileExists,dllPath],
                [If,[StrLength,zbFolName] > 0,
                    [FileExecute,[Var,dllPath],RenameSetNext,zbFolName]
                ]
            ]
        ]
        [IPress,Tool:SubTool:New Folder]
        //MoveUp content subtools into folder
        [If,zbContentCnt > 0,
            [SubToolSelect,zbCurSubIdx+1]
            [Loop,zbContentCnt,
                [IPress,Tool:SubTool:MoveUp]
                [IPress,Tool:SubTool:SelectDown]
            ]
        ]
        //Set folder state AFTER New Folder and MoveUp
        //Use If+arithmetic instead of bitwise |/& for reliability
        [SubToolSelect,zbCurSubIdx]
        [VarSet,zbStatus,[SubToolGetStatus,zbCurSubIdx]]
        //bit0 = 1 (container visible)
        [If,(zbStatus & 1) == 0,
            [VarSet,zbStatus,zbStatus + 1]
        ]
        //bit10 = 0 (clear ClosedFolder for temporary)
        [If,(zbStatus & 1024) == 1024,
            [VarSet,zbStatus,zbStatus - 1024]
        ]
        //bit1 = zbFolEyeVal (folder eye visibility)
        [If,zbFolEyeVal == 1,
            [If,(zbStatus & 2) == 0,
                [VarSet,zbStatus,zbStatus + 2]
            ]
        ,
            [If,(zbStatus & 2) == 2,
                [VarSet,zbStatus,zbStatus - 2]
            ]
        ]
        //bit10 = 1024 if folder should be closed
        [If,zbFolOpenVal != 1,
            [If,(zbStatus & 1024) == 0,
                [VarSet,zbStatus,zbStatus + 1024]
            ]
        ]
        [SubToolSetStatus,zbCurSubIdx,zbStatus]
    ]
]
[MemDelete,zbManMem]
[MemDelete,zbLayerMem]
[VarSet,zbNoteText,[StrMerge,"Import Complete!\n\nEntries processed: ",zbGozCount,"\nErrors: ",zbImportErrors]]
[VarSet,zbNoteText,[StrMerge,zbNoteText,"\nStager stages restored: ",zbStgTotal]]
[If,zbStgSkipped > 0,
[VarSet,zbNoteText,[StrMerge,zbNoteText,"\nStager skipped (SubTools with sculpt layers): ",zbStgSkipped]]
]
[If,[IExists,Tool:Geometry:Home Stage],
,
[VarSet,zbNoteText,[StrMerge,zbNoteText,"\n(This ZBrush has no Stager - stage data ignored)"]]
]
[NoteIButton,"Close"]
[Note,zbNoteText,,,4737096,,300]
]
